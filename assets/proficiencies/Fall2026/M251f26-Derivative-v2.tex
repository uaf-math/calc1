\problem{12}  Compute the derivatives of the following functions.
\begin{subproblems}

\item $\ds f(x)=\frac{x^3-4x^2}{\sec(2x)}$
%\textcolor{red}{type: an obvious quotient}

\vfill
\item $\ds g(x)=x^{-2}(x^3+2x-x^{-\frac{1}{2}})$


\vfill 
\item $\ds h(x) =\ln\left(6+6x^{0.6}\right)$
%\textcolor{red}{type: intelligent use of constants}
\vfill 

\newpage

\item $\ds r(\theta)=\csc\left(\frac{9}{\theta}\right)$
%\textcolor{red}{type: straight chain rule}
\vfill 

%page 2 problems
\item $\ds f(x)=\arctan(2-x)$
%\textcolor{red}{type: arc trig function with product rule}
\vfill
\item $\ds p(t) =\sqrt{16t^3}$
%\textcolor{red}{type: intelligent use of constants}
\vfill 

\newpage
%page 3 problems

\item $\ds f(x)=x^a+b^x$ \hspace{.5cm} where $a$ and $b$ are fixed constants
%\textcolor{red}{type: simple chain with parameter}
\vfill 

\item $\ds g(x)=\sin\left(e^x+\cos(2x)\right)$
%\textcolor{red}{type: straight product rule, e^x, cos x}

\vfill 
\item $\ds y= \frac{\ln{x}}{3+6x}$
%\textcolor{red}{type: double chain, chain with e}

\vfill


\newpage
%page 4 problems
\item $\ds f(x)=\tan \left( \frac{3x}{2-x^2}\right)$
%\textcolor{red}{type: double chain rule, cosine}
\vfill

\item $\ds y=\log_2(x)+\log_2(8)$
%\textcolor{red}{type: easy quotient rule or choice to rewrite, $e^x,$ tempting bad algebra}

\vfill 

\item Find $\ds \frac{dy}{dx}$ for \hspace{.1cm} $\ds x^2e^y=x+y$
%\textcolor{red}{type: implicit differentiation with nontrivial algebra, sine}

\vfill 
\end{subproblems}
